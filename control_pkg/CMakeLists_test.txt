cmake_minimum_required(VERSION 3.8)
project(control_pkg_tests)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

find_package(Eigen3 REQUIRED)
find_package(GTest REQUIRED)

add_library(control_core STATIC
  src/vehicle_model.cpp src/anomaly_detector.cpp src/mpc_controller.cpp)
target_include_directories(control_core PUBLIC ${CMAKE_CURRENT_SOURCE_DIR}/include ${EIGEN3_INCLUDE_DIR})
target_link_libraries(control_core PUBLIC Eigen3::Eigen)

add_executable(test_vehicle_model test/test_vehicle_model.cpp)
target_link_libraries(test_vehicle_model control_core GTest::GTest GTest::Main pthread)

add_executable(test_anomaly_detector test/test_anomaly_detector.cpp)
target_link_libraries(test_anomaly_detector control_core GTest::GTest GTest::Main pthread)

add_executable(test_mpc_controller test/test_mpc_controller.cpp)
target_link_libraries(test_mpc_controller control_core GTest::GTest GTest::Main pthread)

enable_testing()
add_test(NAME test_vehicle_model COMMAND test_vehicle_model)
add_test(NAME test_anomaly_detector COMMAND test_anomaly_detector)
add_test(NAME test_mpc_controller COMMAND test_mpc_controller)
